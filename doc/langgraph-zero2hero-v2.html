"""s10_sqlite_resume.py — 持久化检查点：模型起草审批摘要，进程退出后仍可按 thread_id 恢复审批。

用法（项目根目录，分两次运行，中间可以退出）：
  python code/s10_sqlite_resume.py
第一次输入线程 ID 和退款申请，看到暂停后退出。
第二次再运行，输入同一个线程 ID，按提示输入 y 或 n。
需要模型服务（.env 见 s00_env.md）。检查点写入 code/s10_demo.sqlite，演示完可直接删除。
"""
import os
from pathlib import Path
from typing import Literal

from dotenv import load_dotenv
from langchain_core.runnables import RunnableConfig
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command, interrupt
from pydantic import SecretStr
from typing_extensions import NotRequired, TypedDict

load_dotenv(override=True)
DB_PATH = Path(__file__).with_name("s10_demo.sqlite")


def make_llm(*, temperature: float = 0.0) -> ChatOpenAI:
    return ChatOpenAI(
        model=os.environ["OPENAI_MODEL"],
        api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
        base_url=os.environ["OPENAI_BASE_URL"],
        temperature=temperature,
    )


class State(TypedDict):
    request: str
    summary: NotRequired[str]
    result: NotRequired[str]


def draft(state: State) -> dict:
    # 模型调用单独成节点：它完成后检查点已保存，恢复时不会重新调用（省钱，也避免摘要变化）。
    msg = make_llm().invoke(f"用一句话向审批人概括这条退款申请，不要编造信息：{state['request']}")
    return {"summary": str(msg.content)}


def approval(state: State) -> Command[Literal["done", "cancel"]]:
    # 恢复时本节点会从头重跑，所以 interrupt 之前不要放有副作用的操作。
    ok = interrupt({"question": "是否批准？", "summary": state["summary"]})
    return Command(goto="done" if ok else "cancel")


def done(state: State) -> dict:
    return {"result": "已批准"}


def cancel(state: State) -> dict:
    return {"result": "已拒绝"}


builder = StateGraph(State)
builder.add_node("draft", draft)
builder.add_node("approval", approval)
builder.add_node("done", done)
builder.add_node("cancel", cancel)
builder.add_edge(START, "draft")
builder.add_edge("draft", "approval")
builder.add_edge("done", END)
builder.add_edge("cancel", END)


def ask_yes_no() -> bool:
    while True:
        answer = input("批准退款？输入 y 或 n：").strip().lower()
        if answer in ("y", "n"):
            return answer == "y"
        print("请输入 y 或 n。")


if __name__ == "__main__":
    thread_id = input("线程 ID：").strip()
    if not thread_id:
        raise ValueError("请输入线程 ID")
    config: RunnableConfig = {"configurable": {"thread_id": thread_id}}
    # from_conn_string 是上下文管理器：退出 with 时连接关闭。
    with SqliteSaver.from_conn_string(str(DB_PATH)) as saver:
        app = builder.compile(checkpointer=saver)
        snapshot = app.get_state(config)
        if snapshot.interrupts:
            print("待审批：", snapshot.interrupts[0].value)
            result = app.invoke(Command(resume=ask_yes_no()), config)
        elif isinstance(snapshot.values, dict) and snapshot.values.get("result"):
            raise SystemExit(f"线程 {thread_id} 已经结束：{snapshot.values['result']}。请换一个线程 ID。")
        else:
            text = input("退款申请：").strip()
            if not text:
                raise ValueError("请输入退款申请")
            result = app.invoke({"request": text}, config)
        paused = result.get("__interrupt__")
        print("暂停：" if paused else "结果：", paused[0].value if paused else result["result"])
        print("下一步节点：", app.get_state(config).next)
        if paused:
            print("现在可以退出。下次运行时输入同一个线程 ID，即可继续审批。")

01 / 组件SqliteSaver 把检查点写到文件，换进程仍能读到；thread_id 是找回状态的唯一钥匙。

02 / 逻辑第一次运行时输入申请，模型起草摘要后在 interrupt 暂停并退出。第二次运行输入同一个线程 ID，脚本读到未完成的中断，再让你输入 y/n，并传给 Command(resume=...)。模型调用单独放在 draft 节点，它完成后检查点已保存，恢复时不会重新调用模型。换一个没暂停过的线程 ID，就会当成新申请。

03 / 官方要点interrupt 的 payload 必须可 JSON 序列化；恢复时节点从头重跑到中断处，所以中断前的副作用要幂等。

运行：python code/s10_sqlite_resume.py。第一次输入线程 ID 和退款申请，看到暂停后退出；第二次再运行，输入同一个线程 ID，然后输入 y 或 n。需要模型服务。演示后删除 code/s10_demo.sqlite。单机用 SQLite，多实例部署换 PostgresSaver。

"""s11_refund_prod.py — 退款审批的生产化：把 s09 里「演示可以忽略」的问题逐个补上。

项目根目录运行：python code/s11_refund_prod.py
选一个场景，输入退款申请（可以留空，缺什么会追问，最多 3 轮），审批时输入 y。终端会逐节点打印执行轨迹，
以及每一次网关 / 模型调用，最后对照该场景的预期。
订单库只有已支付的 A100。需要模型服务（.env 见 s00_env.md）和 langgraph>=1.2。支付网关是本文件内的模拟类。

生产里最危险的一步不是调模型，而是「审批通过后调支付网关」：
  1. 网关明确拒绝请求、未提交，稍后恢复，可安全重试；真实 502 不保证未处理；
  2. 网关已扣款但响应超时——调用方不知道钱到底退了没有；
  3. 网关持续不可用，重试用尽——结果未知，只能转对账；
  4. 审批等了几小时，订单在别处已被退掉；
  5. 模型服务自己也会挂，不应拖垮整条流程。
"""
import copy
import os
from typing import Literal

import openai
from dotenv import load_dotenv
from langchain_core.runnables import RunnableConfig
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.errors import NodeError
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command, RetryPolicy, interrupt
from pydantic import BaseModel, Field, SecretStr
from typing_extensions import NotRequired, TypedDict

load_dotenv(override=True)


# ---------- 外部世界：订单库与支付网关（模拟） ----------
ORDERS0 = {"A100": {"order_id": "A100", "amount": 199, "status": "paid"}}
ORDERS = copy.deepcopy(ORDERS0)


class GatewayUnavailable(Exception):
    """模拟网关明确拒绝、确认未提交的请求；真实 502 未必能证明未处理。"""


class GatewayTimeout(Exception):
    """读超时：网关可能已经处理了。重试只有在幂等键保护下才安全。"""


class FakeGateway:
    def reset(self, mode: str) -> None:
        self.mode, self.calls, self.ledger = mode, 0, {}

    def refund(self, key: str, order_id: str, amount: int) -> dict:
        self.calls += 1
        tag = f"      网关 #{self.calls}  key={key}  →"
        if self.mode == "down" or (self.mode == "flaky" and self.calls < 3):
            print(tag, "拒绝，未提交（GatewayUnavailable）")
            raise GatewayUnavailable("请求被明确拒绝，未提交")
        if key in self.ledger:  # 幂等：同一个 key 只生效一次，重复请求返回原结果
            print(tag, "键已存在，返回原结果，不重复退款")
            return self.ledger[key]
        self.ledger[key] = {"order_id": order_id, "amount": amount}
        if self.mode == "timeout_after_commit" and self.calls == 1:
            print(tag, "已入账，但响应超时（GatewayTimeout）")
            raise GatewayTimeout("read timeout")  # 钱已退，响应丢了
        print(tag, f"入账 {amount} 元")
        return self.ledger[key]


GATEWAY = FakeGateway()


# ---------- 模型 ----------
class RefundRequest(BaseModel):
    """提取用户提供的内容，不要猜测未提供的信息。"""

    order_id: str | None = Field(description="订单号；未提供时为 null")
    reason: str | None = Field(description="退款原因；未提供时为 null")


def make_llm(base_url: str | None = None) -> ChatOpenAI:
    # max_retries=0：重试只在图这一层做，避免客户端和图两层叠加成 3×3 次。
    return ChatOpenAI(
        model=os.environ["OPENAI_MODEL"],
        api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
        base_url=base_url or os.environ["OPENAI_BASE_URL"],
        temperature=0,
        max_retries=0,
        timeout=15,

class State(TypedDict):
    user_text: str
    order_id: NotRequired[str]
    reason: NotRequired[str]
    amount: NotRequired[int]
    refund_key: NotRequired[str]
    status: NotRequired[str]
    note: NotRequired[str]
    clarify_turns: NotRequired[int]


CLARIFY_TURNS = 3  # 最多追问三轮，仍不全就结束，避免无限来回


MODEL_URL: str | None = None  # 场景 5 把它指向不可达地址，模拟模型服务宕机


def extract(state: State) -> Command[Literal["lookup", "clarify", "finish"]]:
    text = state["user_text"].strip()
    if text:
        print("      模型调用：抽取订单号和原因")
        req = make_llm(MODEL_URL).with_structured_output(RefundRequest).invoke(text)
    else:
        req = RefundRequest(order_id=None, reason=None)  # 什么都没说，不必调模型，直接追问
    if not isinstance(req, RefundRequest):
        raise TypeError(f"结构化输出类型异常：{type(req).__name__}")  # 代码 bug：不重试，直接暴露
    # 新一轮抽不到时保留上一轮已拿到的字段。
    order_id = (req.order_id or "").strip().upper() or state.get("order_id")
    reason = req.reason or state.get("reason")
    update = {k: v for k, v in (("order_id", order_id), ("reason", reason)) if v}
    if order_id and reason:
        return Command(update=update, goto="lookup")
    # 用户没给全信息是正常业务分支，不是异常：先追问，不进重试也不进 error_handler。
    if state.get("clarify_turns", 0) < CLARIFY_TURNS:
        return Command(update=update, goto="clarify")
    missing = "和".join(name for name, v in (("订单号", order_id), ("退款原因", reason)) if not v)
    return Command(update={**update, "status": "need_input", "note": f"追问 {CLARIFY_TURNS} 轮仍缺{missing}"}, goto="finish")


def clarify(state: State) -> Command[Literal["extract"]]:
    # interrupt 单独放一个节点：恢复时只重跑这里，不会重新调用模型。
    missing = [name for name, key in (("订单号", "order_id"), ("退款原因", "reason")) if not state.get(key)]
    reply = interrupt({"kind": "clarify", "missing": missing, "turn": state.get("clarify_turns", 0) + 1})
    # 把补充内容拼到原文后面，让模型基于完整上下文重新抽取。
    text = "\n".join(part for part in (state["user_text"].strip(), str(reply).strip()) if part)
    return Command(update={"user_text": text, "clarify_turns": state.get("clarify_turns", 0) + 1}, goto="extract")


def lookup(state: State) -> Command[Literal["approval", "finish"]]:
    order = ORDERS.get(state["order_id"])
    if order is None or order["status"] != "paid":
        return Command(update={"status": "not_eligible", "note": "订单不存在或不是已支付"}, goto="finish")
    return Command(update={"amount": order["amount"]}, goto="approval")


def approval(state: State) -> Command[Literal["recheck", "finish"]]:
    # 人可能几小时后才回复。快照只是「审批时看到的样子」，不是执行时的事实。
    ok = interrupt({"kind": "approval", "order_id": state["order_id"], "amount": state["amount"], "reason": state["reason"]})
    if ok is not True:
        return Command(update={"status": "rejected", "note": "审批人拒绝"}, goto="finish")
    return Command(goto="recheck")


def recheck(state: State) -> Command[Literal["charge", "finish"]]:
    # 审批期间订单可能已在别处退掉：先读最新状态；真实系统还要在执行时做原子校验。
    order = ORDERS[state["order_id"]]
    if order["status"] != "paid":
        return Command(update={"status": "stale", "note": "审批期间订单状态已变化，未执行"}, goto="finish")
    # 幂等键必须由业务数据推出。节点恢复时会重跑，若在这里用 uuid4()，每次都是新键，幂等就失效了。
    return Command(update={"refund_key": f"refund-{state['order_id']}"}, goto="charge")


def charge(state: State) -> dict:
    result = GATEWAY.refund(state["refund_key"], state["order_id"], state["amount"])
    ORDERS[state["order_id"]]["status"] = "refunded"
    return {"status": "refunded", "note": f"已退 {result['amount']} 元"}


def charge_unknown(state: State, error: NodeError) -> Command:
    if not isinstance(error.error, GATEWAY_ERRORS):
        raise error.error  # 本地 bug 不冒充「网关异常」
    # 重试用尽。注意：此时「钱没退」和「钱退了但没拿到响应」两种可能都存在，
    # 所以不能标记失败，更不能让用户重新申请。只能挂起等对账。
    return Command(
        update={"status": "refund_unknown", "note": f"网关异常（{type(error.error).__name__}），键 {state['refund_key']}，转人工对账"},
        goto="finish",
    )


def extract_failed(state: State, error: NodeError) -> Command:
    # error_handler 会收到节点的所有异常，不只是 retry_on 里的。
    # 本地 bug（如上面的 TypeError）原样抛出，只有模型服务故障才转人工。
    if not isinstance(error.error, MODEL_ERRORS):
        raise error.error
    # 模型挂了不该让整个退款流程报 500：转人工，把原文留给客服。
    return Command(update={"status": "manual_review", "note": f"模型不可用（{type(error.error).__name__}），原文已留存"}, goto="finish")


def finish(state: State) -> dict:
    return {}


MODEL_ERRORS = (openai.APIConnectionError, openai.APITimeoutError, openai.RateLimitError, openai.InternalServerError)
GATEWAY_ERRORS = (GatewayUnavailable, GatewayTimeout)

builder = (
    StateGraph(State)
    .add_node("extract", extract, retry_policy=RetryPolicy(max_attempts=3, initial_interval=0.1, retry_on=MODEL_ERRORS), error_handler=extract_failed)
    .add_node("clarify", clarify)
    .add_node("lookup", lookup)
    .add_node("approval", approval)
    .add_node("recheck", recheck)
    .add_node("charge", charge, retry_policy=RetryPolicy(max_attempts=3, initial_interval=0.1, retry_on=GATEWAY_ERRORS), error_handler=charge_unknown)

def trace(stream) -> dict | None:
    """逐个打印节点的执行结果；返回中断 payload（没有暂停则为 None）。"""
    paused = None
    for chunk in stream:  # 把流消费完，让本次 invoke 正常收尾、写完检查点
        for node, update in chunk.items():
            if node == "__interrupt__":
                paused = update[0].value
                continue
            name = node.replace("__error_handler__", "") + (" · error_handler" if node.startswith("__error_handler__") else "")
            print(f"  ✓ {name:<24} {update or ''}", flush=True)
    return paused

  1. 查不到资料，模型换个词一直查 → ToolCallLimitMiddleware 限制工具次数；

def parse_skill(path: Path) -> dict[str, str]:
    """读取 SKILL.md：--- 之间是 name / description，后面是正文。"""
    text = path.read_text(encoding="utf-8")
    _, front, body = text.split("---", 2)
    meta = dict(line.split(":", 1) for line in front.strip().splitlines())
    return {"name": meta["name"].strip(), "description": meta["description"].strip(), "body": body.strip()}


SKILLS = {s["name"]: s for s in (parse_skill(p) for p in sorted(SKILLS_DIR.glob("*/SKILL.md")))}


@tool

    """加载一个技能的完整说明。name 必须是「可用技能」目录里的名字。"""
    print(f"      加载技能：{name}")
    skill = SKILLS.get(name)
    if skill is None:
        return f"没有技能 {name!r}，可用：{', '.join(SKILLS)}"
    return skill["body"]


class SkillMiddleware(AgentMiddleware):
    """每次调模型前，把技能目录追加到系统提示词；同时注册 load_skill 工具。"""

    tools = [load_skill]  # 中间件自带的工具，create_agent 会自动加进去

    def __init__(self) -> None:
        catalog = "\n".join(f"- {s['name']}：{s['description']}" for s in SKILLS.values())
        self.addendum = f"\n\n## 可用技能\n{catalog}\n\n回答相关问题前，先用 load_skill 读取对应技能，按技能里的规则回答。"

    def wrap_model_call(self, request: ModelRequest, handler: Callable[[ModelRequest], ModelResponse]) -> ModelResponse:
        blocks = list(request.system_message.content_blocks) if request.system_message else []
        system = SystemMessage(content=blocks + [{"type": "text", "text": self.addendum}])
        return handler(request.override(system_message=system))

from fastmcp.exceptions import ToolError

mcp = FastMCP("refund-desk")

ORDERS = {
    "A100": {"status": "paid", "amount": 199},
    "B200": {"status": "refunded", "amount": 59},
}
KB = {
    "发票": "报销发票抬头为星河科技有限公司，税号 91310115MA1K4QXY2B（演示数据）。",
    "差旅": "差旅住宿标准：一线城市每晚 600 元以内。",

@mcp.tool(annotations={"readOnlyHint": True})
def search_kb(query: str) -> str:
    """在公司制度知识库里按关键词检索；没有命中返回「未找到」。"""
    hits = [doc for term, doc in KB.items() if term in query]
    return "\n".join(hits) if hits else "未找到"


@mcp.tool(annotations={"readOnlyHint": True})
def lookup_order(order_id: str) -> dict:
    """按订单号查询订单状态和金额。"""
    order = ORDERS.get(order_id.strip().upper())
    if order is None:
        # 抛 ToolError → MCP 结果 isError=True → 客户端收到 status="error" 的 ToolMessage，模型能读到原因
        raise ToolError(f"订单 {order_id} 不存在")

def refund_order(order_id: str) -> str:
    """对已支付订单执行退款（模拟）。这是写操作。"""
    order = ORDERS.get(order_id.strip().upper())
    if order is None or order["status"] != "paid":
        raise ToolError(f"订单 {order_id} 不可退款")

工程设计：代码是怎么组织的

s14 把 s04 里和 Agent 写在一起的工具拆成了两个进程：服务端负责「有哪些工具、怎么执行」，客户端负责「怎么用这些工具」。两边只通过 MCP 协议传消息，客户端不 import 服务端的任何代码。读代码时可以按两个问题来看：工具契约由哪一方定义，审批规则又由哪一方定。

模块与职责

模块包含职责边界
s14_mcp_server.py服务端 · 工具的所有者持有数据（ORDERS、KB）；用 @mcp.tool 声明工具，函数签名生成参数 schema，docstring 生成描述；用 annotations 声明只读或有破坏性；业务错误抛 ToolError。不知道调用它的是哪个 Agent、哪个模型
s14_mcp.py客户端 · 工具的使用者拉起并连接服务端；运行时拿到工具目录；把注解翻译成审批策略；组装 Agent；跑对话循环，处理暂停和恢复。只依赖 MCP 协议和工具元数据，不依赖服务端源码

一次调用的完整链路

  1. async with MCPAdapter(SERVER)：参数是 Path，按 stdio 传输处理，拉起 python s14_mcp_server.py 子进程，完成 MCP 握手。stdout 被协议占用，服务端的日志只能走 stderr。
  2. adapter.list_tools()：发送 tools/list，服务端返回每个工具的名字、描述、JSON Schema 和 annotations；适配器把它们转成 LangChain 的 BaseTool，元数据放在 t.metadata。
  3. is_destructive 逐层读取 metadata → mcp → tool → annotations → destructive_hint，得到需要审批的工具名集合 destructive。
  4. create_agent 组装 Agent：模型 + 工具 + HumanInTheLoopMiddleware(interrupt_on=...) + InMemorySaver。审批中间件只拦 destructive 里的工具。
  5. 每轮先记下已有消息数 seen，再 ainvoke。模型想调 refund_order 时，中间件在执行前 interrupt，结果里出现 __interrupt__。
  6. 客户端逐个询问 action_requests，按相同顺序组装 decisions，用 Command(resume={"decisions": ...}) 在同一个 thread_id 上恢复。批准就真正发出 tools/call，拒绝就把拒绝理由作为工具结果交给模型。
  7. 服务端抛出的 ToolError 变成 status="error" 的 ToolMessage。客户端只打印 seen 之后新增的错误，模型自己根据错误原因组织回答。
  8. 退出 async with 时关闭会话并结束子进程，服务端内存里的订单数据随之清空。

设计决策与代价

决策为什么这样做代价 / 生产环境要补的
审批规则读服务端注解，不写死工具名工具由服务端维护。换一个服务端、或者服务端新增了写操作，客户端不用改代码，规则照样生效。注解只是服务端的自我声明。服务端写错或者被篡改，客户端就会放行，所以服务端自己还要校验。
refund_order 在服务端再查一次状态两道防线：客户端审批管「人同不同意」，服务端校验管「这件事能不能做」。审批通过不等于业务上允许。规则有两份；业务规则以服务端为准。
业务失败抛 ToolError，不返回普通字符串MCP 结果带 isError 标记，客户端能把失败和正常结果区分开，可以统计、告警，或者只打印错误。传输层错误（子进程崩溃、连接断开）不会变成 ToolMessage，会直接抛到 ainvoke 外面。
全程 async / ainvoke适配器生成的 MCP 工具只有异步实现，同步 invoke 调不了。入口要 asyncio.run，input() 会阻塞事件循环；在演示里可以接受。
checkpointer 放在客户端 Agent 上interrupt 暂停后要从检查点恢复，和 s08、s10 是同一个机制。示例用 InMemorySaver，进程退出后审批状态就丢了；生产环境换成 s10 的持久化存储。
数据放在服务端进程内存演示时每次拉起子进程都会重置，方便反复跑。生产环境放数据库；如果是多个客户端共享的 HTTP 服务，还要加鉴权。

def is_destructive(t: BaseTool) -> bool:
    """读服务端声明的 destructiveHint；每层都可能缺，用 get 给默认值。"""
    info = t.metadata or dict()
    for key in ("mcp", "tool", "annotations"):
        info = info.get(key) or dict()
    return bool(info.get("destructive_hint", False))

一个 __interrupt__ 里可能有多个 action_requests，因为模型可能一次要求调用多个写工具。decisions 的顺序必须和它们一致。用 while 而不是 if：恢复后模型还可能接着调用下一个写工具。seen 用来只检查本轮新增的消息。

while True:
    question = input("\n问题：").strip()
    if not question:
        break
    seen = len(agent.get_state(config).values.get("messages", []))  # 只检查本轮新增的消息
    out = await agent.ainvoke({"messages": [{"role": "user", "content": question}]}, config)
    while out.get("__interrupt__"):  # 模型想调有破坏性的工具，暂停等人
        decisions = []
        for req in out["__interrupt__"][0].value["action_requests"]:
            ok = input(f"  待审批：{req['name']}({req['args']})，批准吗？(y/n)：").strip().lower() == "y"
            decisions.append({"type": "approve"} if ok else {"type": "reject", "message": "人工拒绝了这次操作"})
        out = await agent.ainvoke(Command(resume={"decisions": decisions}), config)
    for m in out["messages"][seen:]:
        if isinstance(m, ToolMessage) and m.status == "error":
            print(f"  工具返回错误（已交给模型处理）：{m.text}")
    print("回答：", out["messages"][-1].content)

  python code/s15_multi_agent.py supervisor   主从：主 agent 把专家 agent 当工具调用
  python code/s15_multi_agent.py router       并行：先分类，用 Send 同时发给多个专家，再汇总
  python code/s15_multi_agent.py handoff      接力：前台和退款专员互相转接，谁接手谁直接回复用户
自己逐句提问，直接回车退出。需要模型服务（.env 见 s00_env.md）和 langchain>=1.0。

三个专家沿用前几节的模拟数据：invoice（发票）、travel（差旅）、order（订单查询）。
怎么选（官方 Multi-agent 文档的结论）：
  1. 主从：主 agent 统一调度，子 agent 无记忆、上下文隔离；每次都要多一次主 agent 汇总的调用；
  2. 并行：问题能拆成互不依赖的几块时最快，路由本身无状态，多轮对话要外面再包一层；
  3. 接力：要和用户多轮对话、有先后约束（先核实订单再退款）时用；只能串行，不能并行。
同一进程里的协作用这三种就够了；跨服务、跨团队的 agent 互相调用，用 A2A 协议（见教程）。

# ---------- 2. 并行 Router：结构化分类 → Send 扇出 → 汇总 ----------
class Task(BaseModel):
    agent: Literal["invoice", "travel", "order"]
    query: str = Field(description="发给该专员的独立子问题")


class Plan(BaseModel):
    tasks: list[Task] = Field(description="需要咨询的专员；与所有专员无关时为空")


class RouterState(TypedDict):
    question: str
    tasks: NotRequired[list[Task]]
    results: Annotated[list[str], operator.add]  # 多个专家并行写入，用 reducer 合并
    answer: NotRequired[str]


class ExpertInput(TypedDict):
    agent: str
    query: str


planner = model.with_structured_output(Plan)


def classify(state: RouterState) -> dict:
    plan = planner.invoke("把用户问题拆给专员。专员：invoice=发票，travel=差旅，order=订单。\n问题：" + state["question"])
    if not isinstance(plan, Plan):
        raise TypeError(f"分类结果类型不对：{type(plan)}")
    print(f"    分类：{[(t.agent, t.query) for t in plan.tasks]}")
    return {"tasks": plan.tasks}


def fan_out(state: RouterState) -> list[Send] | Literal["synthesize"]:
    if not state["tasks"]:
        return "synthesize"
    # 每个 Send 启动一个 expert 节点实例，它们在同一步里并行执行
    return [Send("expert", {"agent": t.agent, "query": t.query}) for t in state["tasks"]]


def expert(inp: ExpertInput) -> dict:
    return {"results": [f"[{inp['agent']}] {ask_expert(inp['agent'], inp['query'])}"]}


def synthesize(state: RouterState) -> dict:
    if not state["results"]:
        return {"answer": "这个问题不归发票、差旅、订单专员管，请联系 HR。"}
    reply = model.invoke("根据各专员的答复，合成一段回复，不要添加答复里没有的信息。\n问题：" + state["question"]
                           + "\n专员答复：\n" + "\n".join(state["results"]))
    return {"answer": reply.content}


router = (
    StateGraph(RouterState)
    .add_node("classify", classify)
    .add_node("expert", expert)
    .add_node("synthesize", synthesize)
    .add_edge(START, "classify")
    .add_conditional_edges("classify", fan_out, ["expert", "synthesize"])
    .add_edge("expert", "synthesize")
    .add_edge("synthesize", END)
    .compile()
)


# ---------- 3. 接力 Handoffs：平级 agent 用转接工具交出控制权，active_agent 跨轮保留 ----------
class HandoffState(AgentState):
    active_agent: NotRequired[str]


def transfer(to: str, runtime: ToolRuntime) -> Command:
    """通用转接：跳到父图里的另一个 agent 节点，只带「发起转接的 AIMessage + 回执」两条消息。"""
    print(f"    ⇄ 转接给 {to}")
    last_ai = next(m for m in reversed(runtime.state["messages"]) if isinstance(m, AIMessage))
    ack = ToolMessage(content=f"已转接给 {to}", tool_call_id=runtime.tool_call_id)
    return Command(goto=to, update={"active_agent": to, "messages": [last_ai, ack]}, graph=Command.PARENT)

01 / 主从子 agent 包成 @tool，工具 docstring 就是主 agent 选人的依据。主 agent 可以一次发出多个工具调用，所以「住宿标准 + 发票抬头」会同时问两个专员。代价是一次请求固定多一轮主 agent 调用。

02 / 并行和主从的区别在于决定只做一次：结构化输出拆问题，Send("expert", ...) 让同一个节点并发跑多份。并发写同一个 state 键要有 reducer，否则会报 InvalidUpdateError。路由本身无状态，多轮对话时把整张图包成工具交给一个有记忆的 agent。

03 / 接力转接工具返回 Command(goto=..., graph=Command.PARENT) 跳到父图的另一个 agent 节点，只带「发起转接的 AIMessage + ToolMessage 回执」两条消息，保证工具调用成对。父图有 checkpointer，下一轮从 active_agent 继续，所以用户补一句「A100」会直接到退款专员。官方建议能用单 agent + 中间件切换配置时优先用它，更简单。

04 / A2A 协议上面都是同一进程内的协作。agent 部署成独立服务、由不同团队或不同框架实现时，用 A2A（Agent2Agent）：对方发布 Agent Card 描述能力，你用 JSON-RPC 的 SendMessage 发任务、拿回 task 和 artifacts。LangSmith Agent Server 自动为每个 assistant 暴露 /a2a/{assistant_id}。MCP 管「agent 调工具」，A2A 管「agent 调 agent」，在主从模式里远程 A2A agent 也可以包成一个工具。

工程设计：代码是怎么组织的

s15 只有一个文件，分三层：共享层定义一次专家，三种编排层各自组装一个可调用对象，入口层按命令行参数选一个运行。三种模式用的是同一组专家，区别只在编排：谁来决定调用哪个专家、状态存在哪里、谁来回复用户。读代码时可以对比三种模式各自的 State 是什么、控制权在谁手里。

模块与职责

模块包含职责边界
共享层model · ORDERS · lookup_order · EXPERTS · ask_expert三个专家各有自己的提示词和工具；ask_expert 是调用专家的唯一入口，每次传入全新的 messages，专家看不到主对话。三种模式共用，专家代码不会因为编排方式而改
编排 1 · 主从ask_invoice / ask_travel / ask_order · supervisor专家包成工具，主 agent 在 ReAct 循环里自己决定调谁、调几次，最后汇总。状态是主 agent 的 messages，检查点只挂在主 agent 上
编排 2 · 并行Task · Plan · RouterState · ExpertInput · classify / fan_out / expert / synthesize · router一张显式的 StateGraph：先分类拆任务，再用 Send 扇出，最后汇总。分类只做一次，之后不再让模型决定流程。状态是 RouterState，无检查点，每次问答都是独立的
编排 3 · 接力HandoffState · transfer · transfer_to_* · refund_order · 两个 agent · route_handoff · handoff两个平级 agent 作为父图的子图节点，转接工具通过 Command(graph=PARENT) 改变父图的走向。状态是父图的 messages + active_agent，检查点在父图上
入口层if __name__ == "__main__"按 mode 选对象：router 的输入是 question，另外两个的输入是 messages；handoff 每轮打印当前接待人。三个对象的输入和输出形状不同，入口层负责适配

主从 supervisor并行 router接力 handoff

多轮记忆主 agent 的 checkpointer没有，需要外面再包一层父图的 checkpointer + active_agent
适合问题开放、需要边做边判断可拆成互不依赖的子问题，追求延迟和成本可预测需要多轮对话，并且有阶段（先核实再退款）

一次调用的完整链路

  1. 主从：用户问题 → 主 agent 的模型一次发出 ask_travel 和 ask_invoice 两个工具调用 → 每个工具里 ask_expert 用新上下文调用专家 → 两个 ToolMessage 回到主 agent → 主 agent 再调用一次模型，汇总回答。
  2. 并行：START → classify，结构化输出得到 Plan.tasks → fan_out 为每个 task 返回一个 Send("expert", {...}) → 同一步里并发执行多个 expert 实例，每个都向 results 追加一条 → reducer operator.add 合并 → synthesize 汇总 → END。tasks 为空时 fan_out 直接返回 "synthesize"，由它给出转 HR 的兜底回答。
  3. 接力：第一轮 START 读 active_agent（为空）进入 front_desk → 用户说要退款，前台调用 transfer_to_refund → 工具返回 Command(goto="refund_agent", graph=PARENT)，把 active_agent 写进父图 → refund_agent 向用户追问订单号，最后一条是不带工具调用的 AIMessage，route_handoff 返回 END，本轮结束。第二轮用户回复「A100」，START 读到 active_agent=refund_agent，直接进入退款专员。

设计决策与代价

决策为什么这样做代价 / 生产环境要补的
专家只定义一次，所有调用都经过 ask_expert对比三种模式时，变量只有编排方式；上下文隔离和日志也只需要在一个地方处理。专家没有记忆。如果需要专家记住上下文，要在 ask_expert 里为每个会话传入 thread_id。
主从：工具 docstring 就是选人依据主 agent 只能看到工具名和描述，所以描述要写清专家管什么、query 要怎么写（完整、独立的问题）。描述写得不清楚就会选错专家，问题和 s13 的技能描述一样。
并行：Task.agent 用 Literal 约束结构化输出只能从三个专家名里选，后面 EXPERTS[name] 不会因为模型编造了一个名字而 KeyError。新增专家要同时改 Literal、EXPERTS 和分类提示词。
并行：ExpertInput 和 RouterState 分开Send 发给 expert 节点的是私有输入，只有 agent 和 query，节点不需要看全局状态；它的返回值仍写回 RouterState。要多定义一个类型，但每个 expert 实例的输入范围更清楚。
并行：results 必须带 reducer多个 expert 在同一步写同一个键。没有 operator.add 时，LangGraph 不知道怎么合并，会报 InvalidUpdateError。合并顺序不保证，所以 synthesize 只汇总，不依赖顺序。
接力：转接工具只带回两条消息Command.PARENT 跳出子图时，子图内部的消息不会自动带回父图。带上发起转接的 AIMessage 和对应的 ToolMessage 回执，父图里的工具调用才能成对，下一个 agent 调用模型时不会报错。子图里的其他中间消息不会留在父图。
接力：active_agent 存在父图状态里START 的条件边和 route_handoff 都读取它，加上父图的 checkpointer，下一轮会直接回到上次接待的人，实现粘性会话。两个 agent 共享完整的对话历史；如果需要隔离，就要在转接时裁剪消息。

class Task(BaseModel):
    agent: Literal["invoice", "travel", "order"]
    query: str = Field(description="发给该专员的独立子问题")

class Plan(BaseModel):
    tasks: list[Task] = Field(description="需要咨询的专员；与所有专员无关时为空")

class RouterState(TypedDict):
    question: str
    tasks: NotRequired[list[Task]]
    results: Annotated[list[str], operator.add]  # 多个专家并行写入，用 reducer 合并

04 · 并行：分类、扇出、汇总

classify 唯一一次调用模型做决定，返回类型不对就直接报错，不往下猜。fan_out 是条件边函数，可以返回节点名，也可以返回 Send 列表。expert 只返回 results 增量，synthesize 处理空结果的兜底。

s15_multi_agent.py

planner = model.with_structured_output(Plan)

def classify(state: RouterState) -> dict:
    plan = planner.invoke("把用户问题拆给专员。专员：invoice=发票，travel=差旅，order=订单。\n问题：" + state["question"])
    if not isinstance(plan, Plan):
        raise TypeError(f"分类结果类型不对：{type(plan)}")
    print(f"    分类：{[(t.agent, t.query) for t in plan.tasks]}")
    return {"tasks": plan.tasks}

def fan_out(state: RouterState) -> list[Send] | Literal["synthesize"]:
    if not state["tasks"]:
        return "synthesize"
    # 每个 Send 启动一个 expert 节点实例，它们在同一步里并行执行
    return [Send("expert", {"agent": t.agent, "query": t.query}) for t in state["tasks"]]

def expert(inp: ExpertInput) -> dict:
    return {"results": [f"[{inp['agent']}] {ask_expert(inp['agent'], inp['query'])}"]}

def synthesize(state: RouterState) -> dict:
    if not state["results"]:
        return {"answer": "这个问题不归发票、差旅、订单专员管，请联系 HR。"}
    reply = model.invoke("根据各专员的答复，合成一段回复，不要添加答复里没有的信息。\n问题：" + state["question"]
                           + "\n专员答复：\n" + "\n".join(state["results"]))
    return {"answer": reply.content}

06 · 接力：转接就是控制流

HandoffState 在 agent 默认状态上加了 active_agent。transfer 通过 ToolRuntime 拿到当前状态和 tool_call_id，构造回执，返回一个作用于父图的 Command：既跳转，又更新 active_agent 和 messages。

s15_multi_agent.py

class HandoffState(AgentState):
    active_agent: NotRequired[str]

def transfer(to: str, runtime: ToolRuntime) -> Command:
    """通用转接：跳到父图里的另一个 agent 节点，只带「发起转接的 AIMessage + 回执」两条消息。"""
    print(f"    ⇄ 转接给 {to}")
    last_ai = next(m for m in reversed(runtime.state["messages"]) if isinstance(m, AIMessage))
    ack = ToolMessage(content=f"已转接给 {to}", tool_call_id=runtime.tool_call_id)
    return Command(goto=to, update={"active_agent": to, "messages": [last_ai, ack]}, graph=Command.PARENT)

@tool
def transfer_to_refund(runtime: ToolRuntime) -> Command:
    """用户要退款时，转接给退款专员。"""
    return transfer("refund_agent", runtime)

07 · 接力：路由与粘性会话

编译好的 agent 直接作为父图节点。route_handoff 的规则是：最后一条是没有工具调用的 AIMessage，就结束本轮等用户；否则交给 active_agent。转接时它读到的已经是新的 active_agent，和 goto 指向同一个节点，两者不冲突。父图的 checkpointer 让下一轮从 START 直接回到上次的接待人。

s15_multi_agent.py

def route_handoff(state: HandoffState) -> str:
    """最后一条是不带工具调用的 AIMessage → 本轮结束，等用户；否则交给当前 agent。"""
    last = state["messages"][-1]
    if isinstance(last, AIMessage) and not last.tool_calls:
        return END
    return state.get("active_agent") or "front_desk"

handoff = (
    StateGraph(HandoffState)
    .add_node("front_desk", front_desk_agent)  # 编译好的 agent 可以直接当子图节点
    .add_node("refund_agent", refund_agent)
    .add_conditional_edges(START, lambda s: s.get("active_agent") or "front_desk", ["front_desk", "refund_agent"])
    .add_conditional_edges("front_desk", route_handoff, ["front_desk", "refund_agent", END])
    .add_conditional_edges("refund_agent", route_handoff, ["front_desk", "refund_agent", END])
    .compile(checkpointer=InMemorySaver())  # 下一轮从上次的 active_agent 继续
)

async def send(client: Client, text: str, context_id: str) -> str:
    """发一条消息，等 task 结束，按状态取回文字。"""
    request = SendMessageRequest(
        message=Message(message_id=uuid.uuid4().hex, role=Role.ROLE_USER, parts=[new_text_part(text)], context_id=context_id),

        state = TaskState.Name(task.status.state)
        print(f"      ← task {task.id[:8]} {state}")
        if task.status.state == TaskState.TASK_STATE_COMPLETED:

class LangChainExecutor(AgentExecutor):
    """把任意 LangChain agent 接到 A2A：收消息 → 建 task → 调 agent → 写 artifact → 完成。"""

    def __init__(self, name: str, agent) -> None:
        self.name = name
        self.agent = agent

    async def execute(self, context: RequestContext, event_queue: EventQueue) -> None:
        if context.message is None:
            raise ValueError("请求里没有 message")
        task = context.current_task or new_task_from_user_message(context.message)
        if context.current_task is None:
            await event_queue.enqueue_event(task)  # 先登记 task，状态 SUBMITTED
        updater = TaskUpdater(event_queue, task.id, task.context_id)
        user = context.metadata.get("user", "")
        question = context.get_user_input()
        print(f"[{self.name}] user={user!r} context={task.context_id[:8]} 问：{question}")
        if user not in ALLOWED_USERS:
            await updater.reject(updater.new_agent_message([new_text_part(f"用户 {user or '(未提供)'} 无权使用 {self.name}")]))
            return
        await updater.start_work()  # WORKING
        config: RunnableConfig = {"configurable": {"thread_id": task.context_id}}
        out = await self.agent.ainvoke({"messages": [{"role": "user", "content": question}]}, config)
        await updater.add_artifact([new_text_part(out["messages"][-1].content)], name="answer")
        await updater.complete()  # COMPLETED；异常会被框架转成 FAILED

    async def cancel(self, context: RequestContext, event_queue: EventQueue) -> None:
        raise NotImplementedError("演示服务不支持取消")

        return await call_next(request)

01 / 服务端A2A 只规定接口，agent 内部随意。AgentExecutor.execute 是唯一要写的东西：读消息、登记 task、调用自己的 agent、写 artifact、置为完成。这里内部是 create_agent，换成别的框架，调用方完全无感。

02 / 能力发现Agent Card 公开在固定路径，编排方据此生成工具，name 和 description 都不写死。所以 description 和 skills.examples 的质量直接决定会不会被选中，这一点和 s13 的技能描述一样。

03 / 身份令牌证明「谁在调用」（编排服务），metadata.user 说明「替谁调用」（真实员工）。对方按真实用户判断权限，拒绝时返回 REJECTED 和原因，主 agent 如实转告。生产环境应由网关签发、校验令牌，并对用户身份签名，别让调用方随便填。

04 / 多轮与容错context_id 是 A2A 的会话，服务端拿它当 LangGraph 的 thread_id，所以第二问「那如果去成都呢」对方能接上。每次远程调用都加超时，失败时把错误文字交给主 agent，不让整轮崩掉。

工程设计：代码是怎么组织的

s16 把 s15 主从模式里的「专家」换成另一个团队部署的服务。服务端分成三层：协议层（A2A SDK 提供）、适配层（LangChainExecutor，唯一需要自己写的协议代码）、业务层（create_agent）；鉴权是一个 HTTP 中间件。编排方分成四步：发现 → 适配 → 传输 → 编排。和 s15 对照着看，ask_expert 的位置换成了 ask_remote，其余结构基本不变。

模块与职责

模块包含职责边界
s16_a2a_agents.py 协议层create_agent_card_routes · create_jsonrpc_routes · DefaultRequestHandler · InMemoryTaskStoreSDK 提供：公开 Agent Card，解析 JSON-RPC 的 SendMessage，管理 task 的存储和状态流转，把事件变成响应。不需要自己写
s16_a2a_agents.py 适配层LangChainExecutor把一条 A2A 消息翻译成一次 agent 调用：登记 task → 检查身份 → WORKING → ainvoke → 写 artifact → COMPLETED。换框架只改这一层，调用方完全无感
s16_a2a_agents.py 业务层与工厂build_app · APPS · mainbuild_app 用一组参数（端口、名字、描述、维护人、技能、提示词）生成一个完整的服务；APPS 模拟两个团队；main 在一个进程里同时运行两个 uvicorn。两个服务在同一进程里只是为了方便演示，彼此不共享状态
s16_a2a_agents.py 横切BearerAuthPOST 请求没有正确的令牌就返回 401；GET Agent Card 公开，保证能被发现。回答「调用方是谁」，不关心替谁调用
s16_a2a.py 编排方REGISTRY · make_tool · send · main读注册表 → 拉 Agent Card → 每个远程 agent 生成一个工具（闭包里持有 client 和 context_id）→ 交给主 agent。只依赖对方的 Agent Card 和协议，不知道对方内部实现

一次调用的完整链路

  1. 启动：服务端监听 9101、9102。编排方对 REGISTRY 里每个地址 GET /.well-known/agent-card.json，拿到名字、描述、技能、版本、维护人和接口地址。
  2. 适配：make_tool(card, client) 为每个远程 agent 生成一个 StructuredTool。工具名来自 card.name（- 换成 _），描述由 card.description 和 skills 拼成；同时生成一个 context_id，存在闭包里。
  3. 自检：用不带令牌的 httpx 客户端发一次，预期 BearerAuth 返回 401，客户端抛 A2AClientError。
  4. 一次提问：主 agent 的模型同时选中 travel_agent 和 invoice_agent → 每个工具调用 send：构造 SendMessageRequest，message.context_id 用闭包里的 id，metadata.user 填真实用户 → POST JSON-RPC。
  5. 服务端处理：BearerAuth 放行 → DefaultRequestHandler 调用 execute → 新建 task 并入队（SUBMITTED）→ user 不在白名单就 reject（REJECTED，附原因）并返回 → 否则 start_work（WORKING）→ 用 context_id 作为 thread_id 调用 agent → add_artifact → complete（COMPLETED）。execute 里抛出的异常由框架转成 FAILED。
  6. 取结果：streaming=False 时 send_message 只返回最终 task。COMPLETED 就取 artifacts 里的文字；其他状态返回「未完成（状态）：原因」。外面的 wait_for(60) 兜住超时，任何失败都变成工具结果的文字，主 agent 照常汇总。
  7. 追问：用户再问「那如果去成都呢」，主 agent 再次调用 travel_agent，带的是同一个 context_id，服务端的 checkpointer 里有上一轮记录，所以能接上「出差住宿」的上下文。

设计决策与代价

决策为什么这样做代价 / 生产环境要补的
只自己写 AgentExecutor协议细节（JSON-RPC、task 存储、状态机、Agent Card 路由）都交给 SDK，自己只写业务和协议之间的翻译。cancel 没有实现；长任务要支持取消，需要在 execute 里配合检查。
鉴权和授权分两层令牌在 HTTP 中间件里检查（调用方是谁，失败返回 401，请求进不了 agent）；用户权限在 execute 里检查（替谁调用，失败返回 REJECTED 和原因，主 agent 能如实转告）。演示里 metadata.user 可以随便填。生产环境要由网关签名或换成用户令牌，否则谁都能冒充 alice。
Agent Card 公开，SendMessage 需要令牌发现能力不需要权限，调用才需要，这和 MCP 的 list_tools / call_tool 分开是同一个道理。Card 里不要放内部地址和敏感信息。
工具描述来自对方的 Agent Card远程团队改了能力或描述，编排方重新发现即可，不用改代码。选错 agent 的责任落在对方的 description 质量上，所以 Card 要有版本和维护人。
context_id 在 make_tool 闭包里生成编排进程对每个远程 agent 保持一个会话，追问时对方能接上。一个进程只服务一个用户，所以够用。服务多用户时，要按「主 agent 的 thread_id × 远程 agent」维护映射；服务端也应把 user 拼进 thread_id，防止拿到别人的 context_id 后读到别人的对话。
远程失败转成文字，不抛异常一个远程 agent 挂了或者超时，不会让整轮对话崩掉，主 agent 还能用另一个 agent 的结果回答，并说明哪一部分失败了。错误被吞成文字，要靠日志或链路追踪发现；生产环境还要加重试和熔断。
build_app 工厂 + APPS 字典一个服务 = 一组配置，和「每个团队独立部署一份」的真实情况对应；新增 agent 只要加一项。两个服务跑在同一进程只是演示；真实部署时各自独立进程和独立的 task store。

04 · 编排方传输：send 只关心 task 状态

请求里有两类信息：message（内容和 context_id）以及 metadata.user（替谁调用）。返回值按 task 状态分支：只有 COMPLETED 才取 artifacts，其他状态把原因带回去，交给主 agent 决定怎么回复。

s16_a2a.py

async def send(client: Client, text: str, context_id: str) -> str:
    """发一条消息，等 task 结束，按状态取回文字。"""
    request = SendMessageRequest(
        message=Message(message_id=uuid.uuid4().hex, role=Role.ROLE_USER, parts=[new_text_part(text)], context_id=context_id),
        metadata={"user": USER},  # 透传真实发起人
    )
    async for response in client.send_message(request):
        task = response.task
        state = TaskState.Name(task.status.state)
        print(f"      ← task {task.id[:8]} {state}")
        if task.status.state == TaskState.TASK_STATE_COMPLETED:
            return "\n".join(p.text for a in task.artifacts for p in a.parts)
        return f"远程 agent 未完成（{state}）：{get_message_text(task.status.message)}"
    return "远程 agent 没有返回结果"

05 · 编排方适配：Agent Card → 工具

闭包里有三样东西：card（名字和描述）、client（带令牌的连接）、context_id（和这个远程 agent 的会话）。ask_remote 自己负责超时和错误转换，所以主 agent 看到的永远是一个正常返回文字的工具。

s16_a2a.py

def make_tool(card: AgentCard, client: Client) -> StructuredTool:
    """用 Agent Card 生成工具：name、description 都来自对方的自我描述。"""
    context_id = uuid.uuid4().hex  # 本次会话与该 agent 的上下文

    async def ask_remote(query: str) -> str:
        print(f"    → A2A {card.name}：{query}")
        try:
            return await asyncio.wait_for(send(client, query, context_id), timeout=60)
        except (A2AClientError, TimeoutError) as e:
            return f"调用 {card.name} 失败：{type(e).__name__} {e}"

    skills = "；".join(f"{s.name}：{s.description}（例：{'、'.join(s.examples)}）" for s in card.skills)
    return StructuredTool.from_function(
        coroutine=ask_remote,
        name=card.name.replace("-", "_"),
        description=f"{card.description} 技能：{skills}。query 写成完整、独立的问题。",
    )

状态可持久不用 InMemorySaver；每次调用都带稳定的 thread_id
外部写操作先确认渠道的幂等语义与键保留期，再配重试；结果不确定时标「未知」并对账
恢复安全interrupt 前的副作用幂等，或把副作用放到 interrupt 之后的独立节点
模型输出当作不可信输入：缺字段就追问，抽到的订单号以查库为准（s11）
会话成本用官方中间件限制模型和工具调用次数，callback 统计 token；租户硬配额交给网关或计费服务（s12）
专项规则写成技能文件，description 写清何时使用；先实测 token，再决定按需加载还是全量（s13）
外部工具服务MCP 注解只是声明，写操作仍要在客户端审批、在服务端校验；远程服务加鉴权（s14）
多 Agent先确认单 agent 加 Skills 不够用再拆；子 agent 只拿到完成任务所需的上下文；跨服务调用要鉴权、超时，并在链路追踪里串起来（s15）
A2A 平台统一注册中心与网关；令牌鉴权 + 透传真实用户；每次远程调用设超时；限制调用深度防止 agent 互相循环；Agent Card 有版本和负责人（s16）
中断载荷只放可 JSON 序列化的内容
失败分层瞬时 → RetryPolicy；结果未知 → 对账；缺业务信息 → 问用户；未知 bug → 抛出并告警
循环有上限循环先用调用次数中间件设业务上限（s12），再用 recursion_limit 防编排错误；外层处理 GraphRecursionError
等待期间状态变化审批通过后重新读订单；真正执行时仍需原子校验，避免并发竞态（s11）

本篇未涉及：checkpointer 的 durability 模式、长期记忆 store、LangSmith 观测与部署。需要时直接读官方对应页面。
