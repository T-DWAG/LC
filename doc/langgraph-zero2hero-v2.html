"""s10_sqlite_resume.py — 持久化检查点：模型起草审批摘要，进程退出后仍可按 thread_id 恢复审批。

用法（项目根目录，分两次运行，中间进程已退出）：
  python code/s10_sqlite_resume.py start  order-1
  python code/s10_sqlite_resume.py resume order-1 y
需要模型服务（.env 见 s00_env.md）。检查点写入 code/s10_demo.sqlite，演示完可直接删除。
"""
import os
import sys
from pathlib import Path
from typing import Literal

from dotenv import load_dotenv
from langchain_core.runnables import RunnableConfig
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command, interrupt
from pydantic import SecretStr
from typing_extensions import NotRequired, TypedDict

load_dotenv(override=True)
DB_PATH = Path(__file__).with_name("s10_demo.sqlite")


def make_llm(*, temperature: float = 0.0) -> ChatOpenAI:
    return ChatOpenAI(
        model=os.environ["OPENAI_MODEL"],
        api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
        base_url=os.environ["OPENAI_BASE_URL"],
        temperature=temperature,
    )


class State(TypedDict):
    request: str
    summary: NotRequired[str]
    result: NotRequired[str]


def draft(state: State) -> dict:
    # 模型调用单独成节点：它完成后检查点已保存，恢复时不会重新调用（省钱，也避免摘要变化）。
    msg = make_llm().invoke(f"用一句话向审批人概括这条退款申请，不要编造信息：{state['request']}")
    return {"summary": str(msg.content)}


def approval(state: State) -> Command[Literal["done", "cancel"]]:
    # 恢复时本节点会从头重跑，所以 interrupt 之前不要放有副作用的操作。
    ok = interrupt({"question": "是否批准？", "summary": state["summary"]})
    return Command(goto="done" if ok else "cancel")


def done(state: State) -> dict:
    return {"result": "已批准"}


def cancel(state: State) -> dict:
    return {"result": "已拒绝"}


builder = StateGraph(State)
builder.add_node("draft", draft)
builder.add_node("approval", approval)
builder.add_node("done", done)
builder.add_node("cancel", cancel)
builder.add_edge(START, "draft")
builder.add_edge("draft", "approval")
builder.add_edge("done", END)
builder.add_edge("cancel", END)


if __name__ == "__main__":
    mode, thread_id = sys.argv[1], sys.argv[2]
    config: RunnableConfig = {"configurable": {"thread_id": thread_id}}
    # from_conn_string 是上下文管理器：退出 with 时连接关闭。
    with SqliteSaver.from_conn_string(str(DB_PATH)) as saver:
        app = builder.compile(checkpointer=saver)
        if mode == "start":
            result = app.invoke({"request": "订单 A100 商品损坏，申请退款"}, config)
        else:
            result = app.invoke(Command(resume=sys.argv[3] == "y"), config)
        paused = result.get("__interrupt__")
        print("暂停：" if paused else "结果：", paused[0].value if paused else result["result"])
        print("下一步节点：", app.get_state(config).next)

01 / 组件SqliteSaver 把检查点写到文件，换进程仍能读到；thread_id 是找回状态的唯一钥匙。

02 / 逻辑start 先由模型起草摘要，再在 interrupt 暂停并退出；resume 在新进程里用同一 thread_id 传入 Command(resume=...)。模型调用单独放在 draft 节点，它完成后检查点已保存，恢复时不会重新调用模型。传错 thread 则没有可恢复的状态。

03 / 官方要点interrupt 的 payload 必须可 JSON 序列化；恢复时节点从头重跑到中断处，所以中断前的副作用要幂等。

运行：python code/s10_sqlite_resume.py start order-1，再 python code/s10_sqlite_resume.py resume order-1 y · 需要模型服务 · 演示后删除 code/s10_demo.sqlite。单机用 SQLite，多实例部署换 PostgresSaver。

#业务上发生了什么不处理的后果本例的做法
1网关明确拒绝请求且未提交，稍后恢复用户看到「退款失败」，客服收到投诉，其实重试一次就成功RetryPolicy，只重试明确「没处理」的网关错误
2网关已经退款，但响应超时没回来调用方以为失败而重试或让用户重新申请，同一单退两次钱幂等键 refund-订单号，网关按键去重；重试变安全
3网关持续不可用，重试用尽标成「失败」会误导：不知道钱到底退没退error_handler 标 refund_unknown，附幂等键转对账
4审批排了几小时，期间订单已在别处退掉按审批时的快照执行，重复退款审批后重新读取订单状态；执行时仍需原子校验（recheck）
5模型服务宕机或限流整条退款流程报错，用户连申请都提交不了模型节点单独重试，仍失败就转人工并留存原文

再看判断：什么错误能重试

错误类型含义能不能重试怎么处理
GatewayUnavailable网关明确说「没处理」（502、连接被拒）能，无副作用重试
GatewayTimeout不知道网关有没有处理只有带幂等键才能重试 + 幂等键；用尽转对账
ValueError 等本地错误我们自己的 bug 或数据问题不能，重试不会变好让它抛出，被监控发现
interrupt等待人，不是故障不适用不进重试也不进错误处理，照常暂停

↓

↓ 批准

"""s11_refund_prod.py — 退款审批的生产化：把 s09 里「演示可以忽略」的问题逐个补上。

项目根目录运行：python code/s11_refund_prod.py
需要模型服务（.env 见 s00_env.md）和 langgraph>=1.2。支付网关是本文件内的模拟类。

生产里最危险的一步不是调模型，而是「审批通过后调支付网关」：
  1. 网关明确拒绝请求、未提交，稍后恢复，可安全重试；真实 502 不保证未处理；
  2. 网关已扣款但响应超时——调用方不知道钱到底退了没有；
  3. 审批等了几小时，订单在别处已被退掉；
  4. 模型服务自己也会挂，不应拖垮整条流程。
"""
import copy
import os
from typing import Literal

import openai
from dotenv import load_dotenv
from langchain_core.runnables import RunnableConfig
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.errors import NodeError
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command, RetryPolicy, interrupt
from pydantic import BaseModel, Field, SecretStr
from typing_extensions import NotRequired, TypedDict

load_dotenv(override=True)


# ---------- 外部世界：订单库与支付网关（模拟） ----------
ORDERS0 = {"A100": {"order_id": "A100", "amount": 199, "status": "paid"}}
ORDERS = copy.deepcopy(ORDERS0)


class GatewayUnavailable(Exception):
    """模拟网关明确拒绝、确认未提交的请求；真实 502 未必能证明未处理。"""


class GatewayTimeout(Exception):
    """读超时：网关可能已经处理了。重试只有在幂等键保护下才安全。"""


class FakeGateway:
    def reset(self, mode: str) -> None:
        self.mode, self.calls, self.ledger = mode, 0, {}

    def refund(self, key: str, order_id: str, amount: int) -> dict:
        self.calls += 1
        if self.mode == "down" or (self.mode == "flaky" and self.calls < 3):
            raise GatewayUnavailable("请求被明确拒绝，未提交")
        if key in self.ledger:  # 幂等：同一个 key 只生效一次，重复请求返回原结果
            return self.ledger[key]
        self.ledger[key] = {"order_id": order_id, "amount": amount}
        if self.mode == "timeout_after_commit" and self.calls == 1:
            raise GatewayTimeout("read timeout")  # 钱已退，响应丢了
        return self.ledger[key]


GATEWAY = FakeGateway()


# ---------- 模型 ----------
class RefundRequest(BaseModel):
    """提取用户提供的内容，不要猜测未提供的信息。"""

    order_id: str | None = Field(description="订单号；未提供时为 null")
    reason: str | None = Field(description="退款原因；未提供时为 null")


def make_llm(base_url: str | None = None) -> ChatOpenAI:
    # max_retries=0：重试只在图这一层做，避免客户端和图两层叠加成 3×3 次。
    return ChatOpenAI(
        model=os.environ["OPENAI_MODEL"],
        api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
        base_url=base_url or os.environ["OPENAI_BASE_URL"],
        temperature=0,
        max_retries=0,
        timeout=15,
    )


# ---------- 图 ----------
class State(TypedDict):
    user_text: str
    order_id: NotRequired[str]
    reason: NotRequired[str]
    amount: NotRequired[int]
    refund_key: NotRequired[str]
    status: NotRequired[str]
    note: NotRequired[str]


MODEL_URL: str | None = None  # 场景 5 把它指向不可达地址，模拟模型服务宕机


def extract(state: State) -> dict:
    req = make_llm(MODEL_URL).with_structured_output(RefundRequest).invoke(state["user_text"])
    if not isinstance(req, RefundRequest) or not req.order_id or not req.reason:
        raise ValueError("缺少订单号或原因")  # 本地逻辑错误：不重试，直接暴露
    return {"order_id": req.order_id.strip().upper(), "reason": req.reason}


def lookup(state: State) -> Command[Literal["approval", "finish"]]:
    order = ORDERS.get(state["order_id"])
    if order is None or order["status"] != "paid":
        return Command(update={"status": "not_eligible", "note": "订单不存在或不是已支付"}, goto="finish")
    return Command(update={"amount": order["amount"]}, goto="approval")


def approval(state: State) -> Command[Literal["recheck", "finish"]]:
    # 人可能几小时后才回复。快照只是「审批时看到的样子」，不是执行时的事实。
    ok = interrupt({"question": "是否批准退款？", "order_id": state["order_id"], "amount": state["amount"], "reason": state["reason"]})
    if ok is not True:
        return Command(update={"status": "rejected", "note": "审批人拒绝"}, goto="finish")
    return Command(goto="recheck")


def recheck(state: State) -> Command[Literal["charge", "finish"]]:
    # 审批期间订单可能已在别处退掉：先读最新状态；真实系统还要在执行时做原子校验。
    order = ORDERS[state["order_id"]]
    if order["status"] != "paid":
        return Command(update={"status": "stale", "note": "审批期间订单状态已变化，未执行"}, goto="finish")
    # 幂等键必须由业务数据推出。节点恢复时会重跑，若在这里用 uuid4()，每次都是新键，幂等就失效了。
    return Command(update={"refund_key": f"refund-{state['order_id']}"}, goto="charge")


def charge(state: State) -> dict:
    result = GATEWAY.refund(state["refund_key"], state["order_id"], state["amount"])
    ORDERS[state["order_id"]]["status"] = "refunded"
    return {"status": "refunded", "note": f"已退 {result['amount']} 元"}


def charge_unknown(state: State, error: NodeError) -> Command:
    # 重试用尽。注意：此时「钱没退」和「钱退了但没拿到响应」两种可能都存在，
    # 所以不能标记失败，更不能让用户重新申请。只能挂起等对账。
    return Command(
        update={"status": "refund_unknown", "note": f"网关异常（{type(error.error).__name__}），键 {state['refund_key']}，转人工对账"},
        goto="finish",
    )


def extract_failed(state: State, error: NodeError) -> Command:
    # 模型挂了不该让整个退款流程报 500：转人工，把原文留给客服。
    return Command(update={"status": "manual_review", "note": f"模型不可用（{type(error.error).__name__}），原文已留存"}, goto="finish")


def finish(state: State) -> dict:
    return {}


MODEL_ERRORS = (openai.APIConnectionError, openai.APITimeoutError, openai.RateLimitError, openai.InternalServerError)
GATEWAY_ERRORS = (GatewayUnavailable, GatewayTimeout)

builder = (
    StateGraph(State)
    .add_node("extract", extract, retry_policy=RetryPolicy(max_attempts=3, initial_interval=0.1, retry_on=MODEL_ERRORS), error_handler=extract_failed)
    .add_node("lookup", lookup)
    .add_node("approval", approval)
    .add_node("recheck", recheck)
    .add_node("charge", charge, retry_policy=RetryPolicy(max_attempts=3, initial_interval=0.1, retry_on=GATEWAY_ERRORS), error_handler=charge_unknown)
    .add_node("finish", finish)
    .add_edge(START, "extract")
    .add_edge("extract", "lookup")
    .add_edge("charge", "finish")
    .add_edge("finish", END)
)
app = builder.compile(checkpointer=InMemorySaver())


def run(title: str, *, mode: str = "ok", stale: bool = False, model_down: bool = False) -> None:
    global MODEL_URL
    GATEWAY.reset(mode)
    ORDERS.clear()
    ORDERS.update(copy.deepcopy(ORDERS0))
    MODEL_URL = "http://127.0.0.1:9/v1" if model_down else None
    config: RunnableConfig = {"configurable": {"thread_id": title}}
    out = app.invoke({"user_text": "订单 A100 商品损坏，申请退款"}, config)
    if out.get("__interrupt__"):
        if stale:
            ORDERS["A100"]["status"] = "refunded"  # 审批等待期间，别处已经退了
        out = app.invoke(Command(resume=True), config)
    print(f"{title}\n  状态={out['status']}  {out.get('note', '')}\n  网关调用 {GATEWAY.calls} 次，实际到账 {len(GATEWAY.ledger)} 笔")


if __name__ == "__main__":
    run("1 网关明确未提交（重试即可）", mode="flaky")
    run("2 已扣款但响应超时（幂等键兜底）", mode="timeout_after_commit")
    run("3 网关持续不可用（转对账）", mode="down")
    run("4 审批期间订单已被退（执行前复核）", stale=True)
    run("5 模型服务宕机（转人工）", model_down=True)

01 / 先说业务退款里最危险的一步是「审批通过后调支付网关」：它有外部副作用，且失败时结果不确定。上表五个场景都来自这一步及其前后。

02 / 核心取舍重试能不能做，取决于「重复执行会不会再退一次钱」。所以先给操作配幂等键，再谈重试；键必须由业务数据推出（订单号），不能在节点里 uuid4()，因为恢复时节点会重跑。

03 / 失败要分三种状态成功、明确失败、未知。场景 3 里不能写「失败」，否则用户会重新申请；写 refund_unknown 并带上幂等键，由对账确认。

04 / 官方顺序异常 → 看 retry_policy → 用尽才进 error_handler；interrupt 不走这条路。

05 / 模拟与真实的差距网关、订单库是本文件内的模拟类，用来稳定复现故障；真实 502 不代表「一定未处理」；必须确认渠道幂等语义和键保留期。重新读订单只能缩小窗口，仍不能替代执行时的原子校验。

运行：python code/s11_refund_prod.py · 需要模型服务。依次跑五个场景，各打印状态、网关调用次数和实际到账笔数；看场景 2：网关调用 2 次，到账仍是 1 笔。

"""s12_untrusted_output.py — 模型负责抽取，代码负责核对来源与查库。

import re

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langgraph.graph import START, StateGraph
from pydantic import BaseModel, Field, SecretStr
from typing_extensions import NotRequired, TypedDict

load_dotenv(override=True)
ORDERS = {"A100": "paid", "B200": "refunded"}  # 模拟订单库
ORDER_PATTERN = re.compile(r"(?<![A-Za-z0-9])[A-Za-z]\s*-?\s*\d{3}(?!\d)")


class Extracted(BaseModel):
    order_id: str | None = Field(description="用户原话中的订单号；没说则为 null，不得猜测")


llm = ChatOpenAI(
    model=os.environ["OPENAI_MODEL"],
    api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
    base_url=os.environ["OPENAI_BASE_URL"],
    temperature=0,
).with_structured_output(Extracted)


class State(TypedDict):
    text: str
    candidate: NotRequired[str | None]
    order_id: NotRequired[str]
    status: NotRequired[str]


def extract(state: State) -> dict:
    result = llm.invoke(state["text"])
    if not isinstance(result, Extracted):
        raise TypeError("模型未返回预期结构；由调用方监控/处理，不伪装成查无此单")
    return {"candidate": result.order_id}


def canonical(value: str) -> str:
    return re.sub(r"[\s-]", "", value).upper()


def verify(state: State) -> dict:
    # 提取值必须能在原文中找到证据；结构化输出只保证形状，不保证真实。
    evidence = {canonical(m.group()) for m in ORDER_PATTERN.finditer(state["text"])}
    candidate = canonical(state["candidate"]) if state.get("candidate") else None
    if not evidence:
        return {"status": "need_input"}  # 原话没给订单号，继续问模型也没用
    if candidate not in evidence:
        return {"status": "needs_verification"}  # 模型看错或猜错，交人工核对原话
    if candidate not in ORDERS:
        return {"status": "needs_verification"}  # 用户说了也未必是有效订单
    return {"order_id": candidate, "status": "ok"}


graph = (
    StateGraph(State)
    .add_node("extract", extract)
    .add_node("verify", verify)
    .add_edge(START, "extract")
    .add_edge("extract", "verify")
    .compile()
)

if __name__ == "__main__":
    for text in ("我的订单是 a-100，屏幕碎了", "订单 Z999 一直没收到", "帮我退款"):
        out = graph.invoke({"text": text})
        print(f"{text!r} → {out['status']} {out.get('order_id', '')}")

01 / 分工模型负责理解原话；代码规范化并核对原文证据、查询订单库。不成立不是「让模型再试三次」，而是事实待核实。

02 / 安全边界只确认订单存在还不够，真实退款还要验证订单归属与用户身份；本例不做实际退款。

03 / 迁移工单字段、发票号、合同编号等：抽取结果能否用于下游，必须由可核验的业务数据决定。

运行：python code/s12_untrusted_output.py · 三种结果依次为 ok、needs_verification、need_input；需要模型支持结构化输出。

"""s13_step_budget.py — 查不到资料就停止，不把循环次数当成答案质量。

项目根目录运行：python code/s13_step_budget.py
需要模型服务（.env 见 s00_env.md）；知识库工具为模拟。
业务约束：同一问题最多试两个不同检索词；空结果是业务结果，不是让模型无限重试的故障。
"""
import os
from typing import Literal

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command
from pydantic import BaseModel, Field, SecretStr
from typing_extensions import NotRequired, TypedDict

load_dotenv(override=True)
MAX_SEARCHES = 2


class SearchPlan(BaseModel):
    query: str = Field(description="与问题最相关的一个检索词")


llm = ChatOpenAI(
    model=os.environ["OPENAI_MODEL"],
    api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
    base_url=os.environ["OPENAI_BASE_URL"],
    temperature=0,
)
planner = llm.with_structured_output(SearchPlan)


class State(TypedDict):
    question: str
    queries: NotRequired[list[str]]
    hits: NotRequired[list[str]]
    answer: NotRequired[str]


def plan(state: State) -> Command[Literal["search", "finish"]]:
    tried = state.get("queries", [])
    if len(tried) >= MAX_SEARCHES:
        return Command(goto="finish")
    result = planner.invoke(
        f"问题：{state['question']}\n已尝试：{tried}。给一个不同的检索词，不要回答问题。"
    )
    if not isinstance(result, SearchPlan):
        raise TypeError("模型未返回检索词")
    query = result.query.strip()
    if not query or query in tried:
        return Command(goto="finish")  # 换汤不换药，停止浪费检索
    return Command(update={"queries": [*tried, query]}, goto="search")


KB = {"发票": "报销发票抬头为星河科技有限公司。"}  # 模拟授权可见的知识库


def search(state: State) -> Command[Literal["plan", "finish"]]:
    query = state["queries"][-1]
    # 真实系统在这里查库，并核对租户权限；空结果与调用故障必须区分。
    hit = next((doc for term, doc in KB.items() if term in query), None)
    if hit:
        return Command(update={"hits": [hit]}, goto="finish")
    return Command(goto="plan")


def finish(state: State) -> dict:
    if not state.get("hits"):
        # 没证据时不让模型用通用知识编造公司制度；用固定模板回应。
        return {"answer": "知识库未找到这项公司制度，请向 HR 确认。"}
    reply = llm.invoke(f"只根据下面的资料回答问题；资料不足就说不知道。\n资料：{state['hits']}\n问题：{state['question']}")
    return {"answer": str(reply.content)}


graph = (
    StateGraph(State)
    .add_node("plan", plan)
    .add_node("search", search)
    .add_node("finish", finish)
    .add_edge(START, "plan")
    .add_edge("finish", END)
    .compile()
)

if __name__ == "__main__":
    for question in ("公司年假怎么算？", "报销发票抬头是什么？"):
        out = graph.invoke({"question": question}, {"recursion_limit": 20})
        print(question, "实际查询：", out.get("queries", []))
        print("回答：", out["answer"])

01 / 限制与兜底业务上的两次检索上限是主约束；recursion_limit 仅防写错路由导致死循环，不能替代业务预算。

02 / 没有证据不生成这个问题的正确结果可能就是「找不到公司制度」；提示模型不要编造不如根本不让它生成。

03 / 迁移检索、外部 API 查询、自动修复都要分清「未命中」与「临时故障」，分别走结束与可重试路径。

运行：python code/s13_step_budget.py · 需要模型支持结构化输出；年假问题无命中，最多规划两个不同查询词并固定回复；发票问题命中后，模型仅根据资料回答。

"""s14_context_window.py — 对话窗口会丢信息：已核实的事实要独立存储。

项目根目录运行：python code/s14_context_window.py
需模型服务（.env 见 s00_env.md）；订单库为模拟。
真实业务在身份鉴权后才绑定订单；此例仅演示「可见消息」与「业务事实」分离。

import os
import re

from dotenv import load_dotenv
from langchain_core.messages import HumanMessage, SystemMessage
from langchain_core.runnables import RunnableConfig
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.memory import InMemorySaver

from pydantic import SecretStr
from typing_extensions import NotRequired

load_dotenv(override=True)
WINDOW = 4  # 演示按条数；生产按 token 裁剪
ORDERS = {"A100"}  # 模拟查库；真实系统还需验证订单归属
PATTERN = re.compile(r"(?<![A-Za-z0-9])[A-Za-z]-?\d{3}(?!\d)")

    model=os.environ["OPENAI_MODEL"],

    base_url=os.environ["OPENAI_BASE_URL"],

)


class State(MessagesState):
    verified_order: NotRequired[str]
    sent_count: NotRequired[int]


def chat(state: State) -> dict:

        raise TypeError("只接受用户输入")
    # 业务事实由代码查库后写入，不能信任用户或模型自称「已验证」。
    matches = {m.group().replace("-", "").upper() for m in PATTERN.finditer(str(latest.content))}
    found = matches & ORDERS
    verified = next(iter(found)) if len(found) == 1 else state.get("verified_order")
    system = "你是客服；只能确认已核实的订单号，其他信息不知道就说不知道。"

        system += f" 已核实订单号：{verified}。用户询问订单号时直接回答这个已核实的值。"
    recent = state["messages"][-WINDOW:]
    reply = llm.invoke([SystemMessage(content=system), *recent])
    return {"messages": [reply], "sent_count": len(recent), **({"verified_order": verified} if verified else {})}


graph = StateGraph(State).add_node("chat", chat).add_edge(START, "chat").compile(checkpointer=InMemorySaver())


if __name__ == "__main__":
    config: RunnableConfig = {"configurable": {"thread_id": "window-demo"}}
    for text in ("订单 A100 屏幕碎了", "你们几点下班？", "运费谁承担？", "能开发票吗？", "退款多久到账？", "我刚才的订单号是多少？"):
        out = graph.invoke({"messages": [HumanMessage(content=text)]}, config)
    print("保存的历史条数：", len(out["messages"]))
    print("最后一轮发给模型的历史条数：", out["sent_count"])
    print("独立核实的订单号：", out.get("verified_order"))
    print("回答：", out["messages"][-1].content)

"""s15_token_budget.py — 记录实际用量，并阻止下一次超预算会话调用。

项目根目录运行：python code/s15_token_budget.py
需要模型服务（.env 见 s00_env.md）。这里只演示单进程/单线程软限额，
真实的用户/租户限额必须在独立计费服务里原子预留、结算；图状态不是并发配额锁。
"""
import operator
import os
from typing import Annotated, Literal

from dotenv import load_dotenv
from langchain_core.messages import AIMessage, HumanMessage
from langchain_core.runnables import RunnableConfig
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.graph import START, MessagesState, StateGraph
from langgraph.types import Command
from pydantic import SecretStr

load_dotenv(override=True)
BUDGET = 400

llm = ChatOpenAI(
    model=os.environ["OPENAI_MODEL"],
    api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
    base_url=os.environ["OPENAI_BASE_URL"],
    temperature=0,
)


class State(MessagesState):
    tokens_used: Annotated[int, operator.add]


def guard(state: State) -> Command[Literal["chat", "refuse"]]:
    return Command(goto="refuse" if state.get("tokens_used", 0) >= BUDGET else "chat")


def chat(state: State) -> dict:
    # 只发当前一问；避免历史膨胀使本节的「额度」示例变成上下文窗口示例。
    reply = llm.invoke(state["messages"][-1:])
    usage = reply.usage_metadata
    if not usage or not isinstance(usage.get("total_tokens"), int):
        raise RuntimeError("服务端未返回 total_tokens：无法计费，不能悄悄按 0 放行")
    return {"messages": [reply], "tokens_used": usage["total_tokens"]}


def refuse(state: State) -> dict:
    return {"messages": [AIMessage(content="本次会话额度已用完，请联系人工客服。")]}


graph = (
    StateGraph(State)
    .add_node("guard", guard)
    .add_node("chat", chat)
    .add_node("refuse", refuse)
    .add_edge(START, "guard")
    .compile(checkpointer=InMemorySaver())
)

if __name__ == "__main__":
    config: RunnableConfig = {"configurable": {"thread_id": "budget-demo"}}
    for i in range(1, 7):
        out = graph.invoke({"messages": [HumanMessage(content="用三句话介绍退款政策需要考虑的要点。")]}, config)
        print(f"第 {i} 问：累计 {out.get('tokens_used', 0)}/{BUDGET} tokens → {str(out['messages'][-1].content)[:30]}")

02 / 缺数不能当零服务端不返回 usage_metadata.total_tokens 时显式失败；兼容接口必须先验证其用量口径。

03 / 并发不安全InMemorySaver 只为演示；同一用户跨线程、多实例并发必须用外部原子配额服务，不能直接把本节当上线方案。
