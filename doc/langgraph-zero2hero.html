    keys=os.getenv("OPENAI_API_KEY")

        model =os.environ["OPENAI_MODEL"],
        api_key =os.environ["OPENAI_API_KEY"],
        base_url =os.environ["OPENAI_BASE_URL"],
        temperature =temperature,
    )

def add(a:int,b:int) -> int:

    return a + b

def multiply(a:int,b:int) -> int:
    """multiply a and b"""
    return a * b

import os

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langgraph.graph import END, START, StateGraph
from pydantic import SecretStr
from typing_extensions import NotRequired, TypedDict

load_dotenv(override=True)  # 项目 .env 覆盖系统里已有的同名变量


def make_llm(*, temperature: float = 0.0) -> ChatOpenAI:
    if not os.environ["OPENAI_API_KEY"]:
        raise ValueError("OPENAI_API_KEY is not set")
    if not os.environ["OPENAI_BASE_URL"]:
        raise ValueError("OPENAI_BASE_URL is not set")
    if not os.environ["OPENAI_MODEL"]:
        raise ValueError("OPENAI_MODEL is not set")
    return ChatOpenAI(
        model=os.environ["OPENAI_MODEL"],
        temperature=temperature,
        api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
        base_url=os.environ["OPENAI_BASE_URL"],
    )


class State(TypedDict):
    topic: NotRequired[str]
    joke: NotRequired[str]
    polished: NotRequired[str]


def as_text(content: str | list[str | dict]) -> str:
    if not isinstance(content, str):
        raise TypeError(f"expected str, got {type(content).__name__}")
    return content.strip()


llm = make_llm()


def generate_topic(state: State) -> State:
    return {"topic": state.get("topic") or "socks in the dryer"}


def write_joke(state: State) -> State:
    topic = state.get("topic")
    if not topic:
        raise ValueError("topic is not set")
    msg = llm.invoke(f"Write a one-line joke about {topic}. Reply with the joke only.")
    return {"joke": as_text(msg.content)}


def polish(state: State) -> State:
    joke = state.get("joke")
    if not joke:
        raise ValueError("joke is not set")
    msg = llm.invoke(
        "Rewrite this joke so it is shorter and funnier. Reply with the joke only.\n" + joke
    )
    return {"polished": as_text(msg.content)}


graph = StateGraph(State)
graph.add_node("generate_topic", generate_topic)
graph.add_node("write_joke", write_joke)
graph.add_node("polish", polish)
graph.add_edge(START, "generate_topic")
graph.add_edge("generate_topic", "write_joke")
graph.add_edge("write_joke", "polish")
graph.add_edge("polish", END)

result = graph.compile().invoke({"topic":"电梯里的陌生人"})
print("topic:", result["topic"])
print("joke:", result["joke"])
print("polished:", result["polished"])

01 / 组件StateGraph 定义共享状态、节点和边；compile 后才能调用。

02 / 逻辑节点依次更新话题、笑话、润色结果。后两个节点调用模型，本例并非离线图。

运行：python code/s05_min_graph.py

s06_calc_agent.py ↗
START→llm_call→END
有工具请求 ↓　↑ 工具结果

完整代码 · s06_calc_agent.py 绿色底色为重点
import os
import operator  # 导入运算符
from typing import Annotated
from typing_extensions import TypedDict
from dotenv import load_dotenv
from langchain_core.messages import AnyMessage, HumanMessage, SystemMessage
from langchain_core.tools import tool
from langchain_openai import ChatOpenAI
from langgraph.graph import START, StateGraph
from langgraph.prebuilt import ToolNode, tools_condition
from pydantic import SecretStr

load_dotenv(override=True)


def make_llm(*,temperature: float = 0.0) -> ChatOpenAI:
    return ChatOpenAI(
        model =os.environ["OPENAI_MODEL"],
        api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
        base_url =os.environ["OPENAI_BASE_URL"],
        temperature =temperature,
    )



@tool
def add(a:int,b:int) -> int:
    """add a and b"""
    return a + b

@tool
def multiply(a:int,b:int) -> int:
    """multiply a and b"""
    return a * b

@tool
def divide(a:int,b:int) -> float:
    """divide a by b"""
    return a / b

tools = [add, multiply, divide]

model_with_tools = make_llm().bind_tools(tools)

class MessageState(TypedDict):
    messages:Annotated[list[AnyMessage], operator.add]
    llm_calls:int


def llm_call(state:MessageState) :
    return {
        "messages": [
            model_with_tools.invoke(
                [SystemMessage(content="You are a helpful arithmetic assistant.")] + state["messages"]
            )
        ],
        "llm_calls": state["llm_calls"] + 1,
    }


agent_builder = StateGraph(MessageState)
agent_builder.add_node("llm_call", llm_call)
agent_builder.add_node("tools", ToolNode(tools))
agent_builder.add_edge(START, "llm_call")
agent_builder.add_conditional_edges("llm_call", tools_condition)
agent_builder.add_edge("tools", "llm_call")
agent = agent_builder.compile()

init: MessageState = {
    "messages": [HumanMessage(content="Add 3 and 4. Then multiply the result by 2.")],
    "llm_calls": 0,
}
out = agent.invoke(init)
for m in out["messages"]:
    m.pretty_print()
print("llm_calls:", out["llm_calls"])

01 / 组件bind_tools 提供工具定义；ToolNode 执行工具；tools_condition 决定继续还是结束。

02 / 逻辑模型请求工具就执行并回到模型，否则结束。operator.add 让新消息追加到历史，llm_calls 记录模型调用次数。

运行：python code/s06_calc_agent.py · 工具错误、除零或服务端不支持工具调用时可能失败。

完整代码 · s08_hitl.py 绿色底色为重点
from typing import Literal, Optional, TypedDict

from langchain_core.runnables import RunnableConfig
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command, interrupt


# 审批状态：action_details 是待审的动作，status 是还在等、已通过、还是已拒绝。
class ApprovalState(TypedDict):
    action_details: str
    status: Optional[Literal["pending", "approved", "rejected"]]


# 审批节点。interrupt 把问题抛给外面并停住当前这轮。
# 同一个 thread 再次 invoke(Command(resume=...)) 时，resume 的值变成 decision。
# Command(goto=...) 按这个值跳到 proceed 或 cancel，所以这里不用再加普通边。
def approval_node(state: ApprovalState) -> Command[Literal["proceed", "cancel"]]:
    decision = interrupt(
        {
            "question": "Do you approve of this action?",
            "details": state["action_details"],
        }
    )
    return Command(goto="proceed" if decision else "cancel")


# 人点了同意：把状态写成 approved。
def proceed_node(state: ApprovalState):
    return {"status": "approved"}


# 人点了拒绝：把状态写成 rejected。
def cancel_node(state: ApprovalState):
    return {"status": "rejected"}


# interrupt 必须有 checkpointer，挂起的位置才有地方记下来。
# InMemorySaver 只存在当前进程里，按 thread_id 分开每一单。
graph = (
    StateGraph(ApprovalState)
    .add_node("approval", approval_node)
    .add_node("proceed", proceed_node)
    .add_node("cancel", cancel_node)
    .add_edge(START, "approval")
    .add_edge("proceed", END)
    .add_edge("cancel", END)
    .compile(checkpointer=InMemorySaver())
)


# 第一单：转账。第一次 invoke 停在 approval，结果里有 __interrupt__，status 还是 pending。
cfg: RunnableConfig = {"configurable": {"thread_id": "123"}}
paused = graph.invoke(
    {"action_details": "Transfer $500", "status": "pending"},
    config=cfg,
)
print("interrupt", paused.get("__interrupt__"))

# 同一个 thread_id 恢复。resume=True 传回 interrupt()，于是走进 proceed。
approved = graph.invoke(Command(resume=True), cfg)
print("after approve:", approved["status"])








# 第二单用另一个 thread_id，和上面那单的存档互不影响。
cfg2: RunnableConfig = {"configurable": {"thread_id": "456"}}
graph.invoke({"action_details": "Delete DB", "status": "pending"}, cfg2)
# resume=False 传回 interrupt()，于是走进 cancel。
rejected = graph.invoke(Command(resume=False), cfg2)
print("after reject:", rejected["status"])

01 / 组件interrupt 暂停并暴露审批信息；Command 恢复执行或指定下一节点。

02 / 逻辑同一线程传回决定后走批准或拒绝分支。示例模拟人工决定，不执行实际转账或删除操作。

运行：python code/s08_hitl.py · 恢复时节点从头重跑到中断处，之前的副作用应安全可重试。

"""s09_refund_agent.py — 退款申请演示：提取、查单、规则、审批、模拟退款。

项目根目录运行：python code/s09_refund_agent.py
本例使用模拟订单和退款结果，不连接真实支付系统。
"""
import os
from datetime import date, timedelta
from typing import Literal
from uuid import uuid4

from dotenv import load_dotenv
from langchain_core.runnables import RunnableConfig
from langchain_core.tools import tool
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command, interrupt
from pydantic import BaseModel, Field, SecretStr
from typing_extensions import NotRequired, TypedDict

load_dotenv(override=True)


class RefundRequest(BaseModel):
    """提取用户提供的内容，不要猜测未提供的信息。"""

    order_id: str | None = Field(description="订单号；未提供时为 null")
    reason: str | None = Field(description="退款原因；未提供时为 null")


class Order(TypedDict):
    order_id: str
    amount: int
    status: Literal["paid", "refunded", "unpaid"]
    paid_on: str
    simulate_failure: bool


class RefundState(TypedDict):
    user_text: str
    order_id: NotRequired[str | None]
    reason: NotRequired[str | None]
    order: NotRequired[Order | None]
    status: NotRequired[str]
    result: NotRequired[str]


@tool
def lookup_order(order_id: str) -> Order | None:
    """根据订单号查询模拟订单，不存在返回 None。"""
    # 日期相对今天构造，让练习不因日历变化而过期。
    today = date.today()
    orders: dict[str, Order] = {
        "A100": {"order_id": "A100", "amount": 199, "status": "paid", "paid_on": today.isoformat(), "simulate_failure": False},
        "B200": {"order_id": "B200", "amount": 89, "status": "refunded", "paid_on": today.isoformat(), "simulate_failure": False},
        "C300": {"order_id": "C300", "amount": 50, "status": "paid", "paid_on": (today - timedelta(days=31)).isoformat(), "simulate_failure": False},
        "D400": {"order_id": "D400", "amount": 99, "status": "paid", "paid_on": today.isoformat(), "simulate_failure": True},
    }
    return orders.get(order_id.strip().upper())


llm = ChatOpenAI(
    model=os.environ["OPENAI_MODEL"],
    api_key=SecretStr(os.environ["OPENAI_API_KEY"]),
    base_url=os.environ["OPENAI_BASE_URL"],
    temperature=0,
)
extractor = llm.with_structured_output(RefundRequest)


def extract_request(state: RefundState) -> dict:
    request = extractor.invoke(state["user_text"])
    if not isinstance(request, RefundRequest):
        raise TypeError(f"expected RefundRequest, got {type(request).__name__}")
    return {"order_id": request.order_id, "reason": request.reason}


def route_request(state: RefundState) -> Literal["lookup", "missing"]:
    if not (state.get("order_id") or "").strip() or not (state.get("reason") or "").strip():
        return "missing"
    return "lookup"


def missing_info(state: RefundState) -> dict:
    return {"status": "missing_info", "result": "请提供订单号和退款原因后重新申请。"}


def check_order(state: RefundState) -> dict:
    order_id = state.get("order_id")
    if not order_id:
        raise ValueError("order_id is required")
    return {"order": lookup_order.invoke({"order_id": order_id})}


def check_policy(state: RefundState) -> dict:
    """示例规则：订单已支付，且支付日期距今天不超过 30 天。"""

03 · 实现提取节点

def extract_request(state: RefundState) -> dict:
    request = extractor.invoke(state["user_text"])
    if not isinstance(request, RefundRequest):
        raise TypeError(f"expected RefundRequest, got {type(request).__name__}")
    return {"order_id": request.order_id, "reason": request.reason}

04 · 把判断和动作分开

def route_request(state: RefundState) -> Literal["lookup", "missing"]:
    if not (state.get("order_id") or "").strip() or not (state.get("reason") or "").strip():
        return "missing"
    return "lookup"

def missing_info(state: RefundState) -> dict:
    return {"status": "missing_info", "result": "请提供订单号和退款原因后重新申请。"}

05 · 查订单并检查规则

工具查本地订单，节点写入状态；确定性代码判断订单是否存在、已支付且未超过 30 天。模型不决定能否退款。

@tool
def lookup_order(order_id: str) -> Order | None:
    """根据订单号查询模拟订单，不存在返回 None。"""
    # 日期相对今天构造，让练习不因日历变化而过期。
    today = date.today()
    orders: dict[str, Order] = {
        "A100": {"order_id": "A100", "amount": 199, "status": "paid", "paid_on": today.isoformat(), "simulate_failure": False},
        "B200": {"order_id": "B200", "amount": 89, "status": "refunded", "paid_on": today.isoformat(), "simulate_failure": False},
        "C300": {"order_id": "C300", "amount": 50, "status": "paid", "paid_on": (today - timedelta(days=31)).isoformat(), "simulate_failure": False},
        "D400": {"order_id": "D400", "amount": 99, "status": "paid", "paid_on": today.isoformat(), "simulate_failure": True},
    }
    return orders.get(order_id.strip().upper())

def check_order(state: RefundState) -> dict:
